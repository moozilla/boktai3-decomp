// MATCHING_CONCESSION: register-pinning
#include "global.h"
#define MotionEffectPool_StartTrail sub_081F2A08
struct TrailCamera {
    u8 pad0[0x44];
    s32 x;
};
struct TrailSprite {
    u32 flags;
    u8 pad4[0x18];
    s16 x, y, z;
    u8 pad22[10];
};
struct MotionTrail {
    u8 active, delay;
    u8 pad2[2];
    s32 position[4];
    s32 trail[4][4];
    s32 velocity[3];
    s32 pad60;
    s32 target[4];
    s32 style, paramA, paramB, pad80, pad84;
    u8 pad88[0x108 - 0x88];
    struct TrailSprite sprites[7];
    u8 pad23c[4];
};
extern struct TrailCamera *gUnk_020004B8;
u8 *MotionEffectPool_FindFreeEntry(u8 *, s32 *);
void Vector3_CopyComponents(void *, void *);
void Sprite_LinkToList(void *, int);
void AnimData_SelectPaletteBank(void *, s32);
void SpriteTrail_InitRenderCollider(void *);
void SpriteTrail_InitCollider(void *);
s32 MotionEffectPool_StartTrail(u8 *pool, s32 *from, s32 *to, u32 style, s32 paramA, s32 paramB)
{
    s32 index;
    register struct MotionTrail *e asm("r6");
    s32 i;
    struct TrailSprite *s;
    struct TrailSprite *t;
    struct TrailSprite *s2;
    struct TrailSprite *t2;
    s32 *trail;
    register s32 *pos asm("r5");
    register s32 *to2 asm("r7") = to;
    e = (struct MotionTrail *)MotionEffectPool_FindFreeEntry(pool, &index);
    if (e == 0)
        return -1;
    e->delay = 0;
    e->active = 1;
    e->pad84 = 0;
    pos = e->position;
    Vector3_CopyComponents(pos, from);
    Vector3_CopyComponents(e->target, to2);
    e->style = style;
    e->velocity[0] = (to2[0] - from[0]) >> 4;
    e->velocity[1] = (to2[1] - from[1]) >> 4;
    e->velocity[2] = (to2[2] - from[2]) >> 4;
    e->paramA = paramA;
    e->paramB = paramB;
    e->pad80 = 0;
    s = e->sprites;
    t = e->sprites + 4;
    s2 = s;
    trail = e->trail[0];
    for (i = 3; i >= 0; i--, s2++, s++) {
        Vector3_CopyComponents(trail, pos);
        s2->x = (e->position[0] >> 12) - gUnk_020004B8->x;
        s2->y = ((pos[2] + pos[1]) >> 12) + 0x60;
        s2->z = pos[2] >> 12;
        s2->flags &= ~1;
        Sprite_LinkToList(s, 0);
        trail += 4;
    }
    t2 = t;
    for (i = 2; i >= 0; i--, t2++, t++) {
        t2->x = (e->position[0] >> 12) - gUnk_020004B8->x;
        t2->y = ((pos[2] + pos[1]) >> 12) + 0x60;
        t2->z = pos[2] >> 12;
        t2->flags &= ~1;
        Sprite_LinkToList(t, 0);
    }
    AnimData_SelectPaletteBank((u8 *)e + 0x23c, 0x1e5);
    SpriteTrail_InitRenderCollider(e);
    SpriteTrail_InitCollider(e);
    *(u32 *)(pool + 0x18) |= 1 << index;
    return 0;
}
